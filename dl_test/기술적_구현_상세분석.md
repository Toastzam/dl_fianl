## 💻 핵심 코드 구현 분석

### 1. 상태 관리 패턴

```jsx
// DogSimilaritySearch.jsx - 중앙 집중식 상태 관리
function DogSimilaritySearch() {
  // 핵심 상태들
  const [currentPage, setCurrentPage] = useState('search');
  const [searchResults, setSearchResults] = useState(null);
  const [selectedDog, setSelectedDog] = useState(null);
  const [originalImage, setOriginalImage] = useState(null);
  const [queryKeypointImage, setQueryKeypointImage] = useState(null);

  // 상태 전환 핸들러들
  const handleSearchResults = (results, originalImg, queryKeypointImg) => {
    // 모든 관련 데이터를 한번에 업데이트
    setSearchResults(results);
    setOriginalImage(originalImg);
    setQueryKeypointImage(queryKeypointImg);
    setCurrentPage('gallery'); // 다음 단계로 이동
  };

  const handleSelectDog = (dog) => {
    setSelectedDog(dog);
    setCurrentPage('detail');
  };

  // 상태 초기화 (중요!)
  const handleBackToSearch = () => {
    setCurrentPage('search');
    setSearchResults(null);
    setSelectedDog(null);
    setOriginalImage(null);
    setQueryKeypointImage(null);
  };
}
```

**학습 포인트**: 
- 관련된 상태들을 함께 업데이트하여 일관성 유지
- 명확한 상태 초기화로 사이드 이펙트 방지
- 단방향 데이터 플로우 유지

### 2. API 통신 및 에러 핸들링

```jsx
// SearchChatbotModal.jsx - 견고한 API 통신
const handleSearch = async () => {
  if (!selectedFile) return;

  setIsLoading(true);
  setError(null);

  try {
    // FormData 생성 (파일 업로드용)
    const formData = new FormData();
    formData.append('image', selectedFile);

    const response = await fetch('http://localhost:8000/search-similar', {
      method: 'POST',
      body: formData,
    });

    if (!response.ok) {
      throw new Error(`HTTP error! status: ${response.status}`);
    }

    const data = await response.json();
    
    // 성공 시 부모 컴포넌트에 결과 전달
    onSearchResults(
      data.similar_dogs,
      URL.createObjectURL(selectedFile), // 원본 이미지 URL 생성
      data.query_keypoint_image
    );

  } catch (error) {
    console.error('검색 중 오류 발생:', error);
    setError('검색 중 오류가 발생했습니다. 다시 시도해주세요.');
  } finally {
    setIsLoading(false);
  }
};
```

**학습 포인트**:
- try-catch-finally 패턴으로 완전한 에러 핸들링
- 로딩 상태 관리로 사용자 경험 개선
- FormData 사용한 파일 업로드 구현

function SearchPetPage({ searchResults, onSelectDog, onBackToSearch, originalImage }) {

          </button>
        </div>
      </div>

      {/* 결과 그리드 */}
      <div className="p-6">
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-6">
          {searchResults?.map((dog, index) => (
            <div 
              key={`${dog.filename}-${index}`}
              className="bg-white rounded-lg shadow-lg overflow-hidden hover:shadow-xl transition-shadow cursor-pointer"
              onClick={() => onSelectDog(dog)}
            >
              <div className="relative">
                {/* 로딩 스피너 */}

                  <div className="absolute inset-0 flex items-center justify-center bg-gray-100">
                    <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-600"></div>
                  </div>
                )}
                
                {/* 실제 이미지 */}
                <img
                  src={`http://localhost:8000/uploads/${dog.filename}`}
                  alt={`유사한 강아지 ${index + 1}`}
                  className="w-full h-48 object-cover"
                  onLoad={() => handleImageLoad(dog.filename)}
                />

                {/* 유사도 배지 */}
                <div className="absolute top-2 right-2 bg-black/70 text-white px-2 py-1 rounded text-sm">
                  {(dog.similarity * 100).toFixed(1)}%
                </div>
              </div>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}

### 4. Canvas 기반 시각화

```jsx
// FocusedImage.js - Canvas API 활용한 포커스 효과
import React, { useRef, useEffect } from 'react';

function FocusedImage({ src, focusX, focusY, radius, width, height }) {
  const canvasRef = useRef(null);
  const imageRef = useRef(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    const image = imageRef.current;
    
    if (!canvas || !image) return;

    const ctx = canvas.getContext('2d');
    
    const drawFocusEffect = () => {
      // 캔버스 크기 설정
      canvas.width = width;
      canvas.height = height;

      // 전체 영역을 어둡게 처리
      ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
      ctx.fillRect(0, 0, width, height);

      // 포커스 영역을 밝게 만들기 (destination-out 블렌딩 사용)
      ctx.globalCompositeOperation = 'destination-out';
      ctx.beginPath();
      ctx.arc(focusX, focusY, radius, 0, 2 * Math.PI);
      ctx.fill();

      // 블렌딩 모드 복원
      ctx.globalCompositeOperation = 'source-over';
    };

    // 이미지 로드 완료 시 효과 적용
    if (image.complete) {
      drawFocusEffect();
    } else {
      image.onload = drawFocusEffect;
    }

  }, [src, focusX, focusY, radius, width, height]);

  return (
    <div style={{ position: 'relative', width, height }}>
      {/* 원본 이미지 */}
      <img
        ref={imageRef}
        src={src}
        alt="Original"
        style={{ width: '100%', height: '100%', objectFit: 'cover' }}
      />
      
      {/* 포커스 효과 캔버스 */}
      <canvas
        ref={canvasRef}
        style={{
          position: 'absolute',
          top: 0,
          left: 0,
          width: '100%',
          height: '100%',
          pointerEvents: 'none',
        }}
      />
    </div>
  );
}

export default FocusedImage;
```

**학습 포인트**:
- Canvas API의 Composite Operation 활용
- useRef를 통한 DOM 요소 직접 조작
- 이미지 로딩 상태 감지 및 처리
